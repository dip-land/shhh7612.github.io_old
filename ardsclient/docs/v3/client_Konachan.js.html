    'use strict';

const BaseHandler = require("./handlers/BaseHandler");

/**
 * Gets a post from konachan.
 * @example new Konachan({ limit: 50, fullPost: false, tags: ['tag1', 'tag2']}).post.then(data => {console.log(data)})
*/
class Konachan {
    /**
     * @typedef {Object} KonachanOptions                 Represents the possible options for konachan
     * @property {boolean} [fullPost = false]            If true will return the full post
     * @property {number} [limit = 50]                   Number of posts to fetch
     * @property {string[]} [tags]                       Array of tags you want to search for
    */
    /**
     * Gets a post from konachan.
     * @param {KonachanOptions} [options = {}]           Represents the possible options for konachan
    */
    constructor(options = {}) {
        this.limit = options.limit ?? 50;
        this.fullPost = options.fullPost ?? false;
        this.tags = `&tags=${options.tags}`;
        /**
         * possible ways to order the posts
         * @type {string[]}
        */
        const order = ['id', 'id_desc', 'score', 'score_asc', 'mpixels', 'mpixels_asc', 'landscape', 'portrait']
        /**
         * url to konachan
         * @type {string}
        */
        const url = `https://konachan.com/post.json?limit=${this.limit}${this.tags}&order=${order[Math.floor(Math.random() * order.length)]}`;
        /**
         * Chosen post's resolveable data
         * @type {Promise<KonachanPost>}
        */
        this.post = new BaseHandler().getPayload(url, "konachan", this.fullPost);
    }
}
module.exports = Konachan;
