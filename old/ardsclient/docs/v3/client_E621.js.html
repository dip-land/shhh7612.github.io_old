    'use strict';

const BaseHandler = require("./handlers/BaseHandler");

/**
 * Gets a post from E621. 
 * @see https://e621.net/help/api 
 * for e621 api help and ratelimits.
 * @example new E621({ limit: 50, fullPost: false, tags: ['tag1', 'tag2']}, userAgent: 'MyProject/1.0 (by username on e621)').post.then(data => {console.log(data)})
*/
class E621 {
    /**
     * @typedef {Object} E621Options                     Represents the possible options for E621.
     * @property {boolean} [fullPost = false]            If true will return the raw/full post data.
     * @property {number} [limit = 50]                   Number of posts to fetch.
     * @property {string[]} [tags]                       Array of tags you want to search for.
     * @property {string} userAgent                      Descriptive User-Agent for your project.  
     *                                                   *e.g. MyProject/1.0 (by username on e621)*
    */
    /**
     * Gets a post from E621. 
     * @see https://e621.net/help/api                    for e621 api help and ratelimits.
     * @param {E621Options} [options = {}]               Represents the possible options for E621.
    */
    constructor(options = {}) {
        this.limit = options.limit ?? 100;
        this.fullPost = options.fullPost ?? false;
        this.tags = `&tags=${options.tags}`;
        /**
         * URL to E621.
         * @type {string}
        */
        //&page=none
        let url = `https://e621.net/posts.json?limit=${this.limit}${this.tags}`;
        /**
         * Chosen post's resolveable data.
         * @type {Promise<E621Post>}
        */
        this.post = new BaseHandler().getPayload(url, "E621", this.fullPost, this.userAgent);
    }
}
