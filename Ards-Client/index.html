Installation

Node.js 12.0.0 or newer is required to use this package.
//for npm
npm i ards-client
                
//for yarn
yarn add ards-client

Basic Usage

Replace the class with the one you want to use, list can be found here. Replace the function you want to use with a valid function from a class of your choice, some functions can take arguments.
const {Client} = require("ards-client");
new Client().class.function().then(request => {
    console.log(request);
}).catch(error => {
    console.log(error);
});

//or

const ards = require("ards-client"), ardsClient = new ards.Client();
ardsClient.class.function().then(request => {
    console.log(request);
}).catch(error => {
    console.log(error);
});

Example Usage

This example is for the Rule34 function using Discord.js and it's embeds.
const ards = require("ards-client"), ardsClient = new ards.Client(), {MessageEmbed} = require('discord.js');
tags = ['solo_focus', 'rating:explicit']
ardsClient.hentai.rule34(tags).then(request => {
    if(request.post.failed){
        return message.channel.send(new MessageEmbed()
        .setDescription(`${tags.join(', ')} is not a valid tag or there was an error with that command.  For list of avalible tags [click here](https://rule34.xxx/index.php?page=tags&s=list).`)
        .setColor('#ff0000'))
    }
    message.channel.send(new MessageEmbed()
    .setImage(request.post.file_url)
    .setDescription(`Image not loading? [Click Here](${request.post.file_url}})\n[View on Rule34](https://rule34.xxx/index.php?page=post&s=view&id=${request.post.id})`)
    .setFooter(`👍 ${request.post.score} | ${new Date(request.post.created_at).toLocaleString()}`))
}).catch(error => {
    console.log(error);
});
